pwd — показать текущую папку

ls — показать файлы и папки

ls -la — подробный список, включая скрытые файлы

cd — перейти в другую папку

cd .. — подняться на уровень выше

cd ~ — перейти в домашнюю папку

mkdir — создать папку

touch — создать пустой файл

cp — скопировать файл или папку

mv — переместить или переименовать файл/папку

rm — удалить файл

rm -r — удалить папку с содержимым

cat — вывести содержимое файла

less — открыть файл для просмотра с прокруткой

head — показать начало файла

tail — показать конец файла

tail -f — следить за файлом в реальном времени

nano — открыть простой текстовый редактор

grep — найти строки с нужным текстом

grep -r — искать текст по всем файлам в папке

find — искать файлы и папки

wc -l — посчитать количество строк

| — передать результат одной команды в другую

> — записать вывод команды в файл

>> — добавить вывод команды в конец файла

chmod — изменить права доступа к файлу

chown — изменить владельца файла

sudo — выполнить команду от администратора

apt update — обновить список пакетов

apt upgrade — обновить установленные пакеты

apt install — установить программу

apt remove — удалить программу

ps aux — показать запущенные процессы

top — мониторинг CPU и RAM

kill — завершить процесс по его ID

kill -9 — принудительно завершить процесс

df -h — показать свободное место на дисках

du -sh — показать размер папки

free -h — показать использование оперативной памяти

whoami — показать текущего пользователя

uname -a — показать информацию о системе

man — открыть справку по команде

history — показать историю введённых команд

clear — очистить экран терминала
